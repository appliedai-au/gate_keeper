=== y1-bun-02-job-queue|Bun SQLite job queue|typescript|src/queue.ts
CONTEXT: Background job queue on bun:sqlite shared by several worker processes; each job must be claimed by exactly one worker. A job runs at most MAX_ATTEMPTS times in total, then becomes dead. Dead jobs must be kept for inspection; purgeOld only removes done jobs. All times are milliseconds.

import { Database } from "bun:sqlite";

db.run("PRAGMA journal_mode = WAL");
const MAX_ATTEMPTS = 3;

type Job = { id: number; kind: string; payload: string; attempts: number };
type Handler = (payload: unknown) => Promise<void>;

const pick = db.query("SELECT id, kind, payload, attempts FROM jobs WHERE status = 'queued' AND run_at <= ? ORDER BY run_at LIMIT 1");
const markRunning = db.query("UPDATE jobs SET status = 'running', attempts = attempts + 1 WHERE id = ?");
const markDone = db.query("UPDATE jobs SET status = 'done', finished_at = ? WHERE id = ?");
const reschedule = db.query("UPDATE jobs SET status = 'queued', run_at = ?, last_error = ? WHERE id = ?");
const markDead = db.query("UPDATE jobs SET status = 'dead', finished_at = ?, last_error = ? WHERE id = ?");

export function claim(): Job | null {
  const job = pick.get(Date.now()) as Job | null;
  if (!job) return null;
  markRunning.run(job.id); ⟦BUG concurrency|select and update are separate statements outside a transaction, so two worker processes can claim the same job; use BEGIN IMMEDIATE or an UPDATE ... WHERE status = 'queued' RETURNING⟧
  return job;
}

export async function runOne(handlers: Map<string, Handler>) {
  const job = claim();
  if (!job) return false;
  const handler = handlers.get(job.kind);
  if (!handler) { markDead.run(Date.now(), `no handler for ${job.kind}`, job.id); return true; } ⟦OK⟧
  try {
    await handler(JSON.parse(job.payload));
    markDone.run(Date.now(), job.id);
  } catch (err) {
    const attempts = job.attempts + 1;
    if (attempts > MAX_ATTEMPTS) { ⟦BUG correctness|attempts already counts this run, so > lets a job run MAX_ATTEMPTS + 1 times; must be >=⟧
      markDead.run(Date.now(), String(err), job.id);
    } else {
      reschedule.run(Date.now() + 1000 * 2 ** attempts, String(err), job.id);
    }
  }
  return true;
}

export function purgeOld(olderThanMs: number) {
  return db.query("DELETE FROM jobs WHERE finished_at < ?").run(Date.now() - olderThanMs).changes; ⟦BUG data-integrity|no status = 'done' filter, so dead jobs (which also get finished_at) are deleted too⟧
}

export async function workLoop(handlers: Map<string, Handler>, signal: AbortSignal) {
  while (!signal.aborted) {
    const worked = await runOne(handlers);
    if (!worked) await Bun.sleep(500);
  }
}

import { redis } from "bun";
import { sendSms } from "./sms";

const TTL_SECONDS = 300;
const MAX_TRIES = 5;

function normalize(phone: string) {
  const p = phone.replace(/[^\d+]/g, "");
  if (!/^\+\d{8,15}$/.test(p)) throw new Error("invalid phone");
  return p;
}

function sixDigits() {
  const n = crypto.getRandomValues(new Uint32Array(1))[0] % 1_000_000; ⟦OK⟧
  return n.toString().padStart(6, "0");
}

export async function requestCode(phone: string) {
  const p = normalize(phone);
  const code = sixDigits();
  await redis.set(`otp:${p}`, code);
  await redis.expire(`otp:${p}`, TTL_SECONDS * 1000); ⟦BUG correctness|EXPIRE takes seconds; multiplying by 1000 keeps codes alive ~3.5 days instead of 5 minutes⟧
  await redis.expire(`otp-tries:${p}`, TTL_SECONDS);

}

export async function verifyCode(phone: string, input: string) {
  const p = normalize(phone);
  const tries = await redis.incr(`otp-tries:${p}`);
  if (tries === 1) await redis.expire(`otp-tries:${p}`, TTL_SECONDS);
  if (tries >= MAX_TRIES) return { ok: false, reason: "locked" }; ⟦BUG correctness|>= locks on the 5th attempt, so only 4 attempts are allowed; must be > MAX_TRIES⟧
  const stored = await redis.get(`otp:${p}`);
  if (stored === null) return { ok: false, reason: "expired" };
  if (stored !== input) return { ok: false, reason: "wrong" };
  return { ok: true }; ⟦BUG security|the code is never deleted after a successful check, so it can be replayed until it expires (must be single-use)⟧
}

    url.searchParams.set("client_id", CLIENT_ID);
    url.searchParams.set("state", state);
