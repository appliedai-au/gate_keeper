CONTEXT: Per-client token-bucket rate limiter shared across threads. Each bucket holds at most `capacity` tokens and refills at `refill_per_sec` tokens per second; a request costs one whole token. Unknown clients start with a full bucket. `prune` removes buckets that have been idle for more than `idle_secs`.

use std::collections::HashMap;
use std::sync::Mutex;
use std::time::Instant;

pub struct Bucket {
    tokens: f64,
    last: Instant,
}

pub struct Limiter {
    capacity: f64,
    refill_per_sec: f64,
    buckets: Mutex<HashMap<String, Bucket>>,
}

impl Limiter {
    pub fn new(capacity: u32, refill_per_sec: u32) -> Self {
        Limiter {
            capacity: capacity as f64,
            refill_per_sec: refill_per_sec as f64,
            buckets: Mutex::new(HashMap::new()),
        }
    }

    pub fn allow(&self, client: &str) -> bool {
        let mut map = self.buckets.lock().unwrap(); ⟦OK⟧
        let now = Instant::now();
        let bucket = map.entry(client.to_string()).or_insert(Bucket { tokens: 0.0, last: now }); ⟦BUG correctness|new clients must start with a full bucket, but tokens starts at 0 so their first request is rejected⟧
        let last = bucket.last;
        let elapsed = now.duration_since(last).as_millis() as f64; ⟦BUG correctness|elapsed is measured in milliseconds but the rate is per second, so buckets refill 1000x too fast⟧
        bucket.last = now;
        bucket.tokens += elapsed * self.refill_per_sec; ⟦BUG correctness|tokens are never capped at capacity, so an idle client accumulates an unlimited burst⟧

        if bucket.tokens > 0.0 { ⟦BUG correctness|a request costs a whole token; with > 0.0 a fractional balance admits the request and drives tokens negative⟧
            bucket.tokens -= 1.0;
            return true;
        }
        false
    }

    pub fn remaining(&self, client: &str) -> u32 {
        let map = self.buckets.lock().unwrap();
        map.get(client).map(|b| b.tokens as u32).unwrap_or(self.capacity as u32) ⟦OK⟧
    }

    pub fn prune(&self, idle_secs: u64) {
        let mut map = self.buckets.lock().unwrap();
        let now = Instant::now();
        map.retain(|_, b| now.duration_since(b.last).as_secs() > idle_secs); ⟦BUG correctness|retain keeps the entries for which the closure is true, so this keeps idle buckets and deletes the active ones⟧
    }
}

=== y1-rust-03-lru-cache|Rust LRU page cache|rust|src/lru.rs
CONTEXT: Fixed-capacity LRU cache for rendered pages. It must never hold more than `capacity` entries. `get` marks an entry as most recently used and `peek` does not. Inserting a new key into a full cache evicts the least recently used entry; overwriting an existing key must not evict anything. `shrink_to` lowers the capacity and evicts down to it.
---
use std::collections::{HashMap, VecDeque};

pub struct LruCache {
    capacity: usize,
    map: HashMap<String, Vec<u8>>,
    order: VecDeque<String>,
}

impl LruCache {
    pub fn new(capacity: usize) -> Self {
        assert!(capacity > 0, "capacity must be positive"); ⟦OK⟧
        LruCache { capacity, map: HashMap::with_capacity(capacity), order: VecDeque::new() }
    }

    fn touch(&mut self, key: &str) {
        if let Some(pos) = self.order.iter().position(|k| k == key) {
            self.order.remove(pos);
        }
        self.order.push_back(key.to_string());
    }

    pub fn get(&mut self, key: &str) -> Option<&Vec<u8>> {
        if self.map.contains_key(key) {
            self.touch(key);
        }
        self.map.get(key)
    }

    pub fn peek(&self, key: &str) -> Option<&Vec<u8>> {
        self.map.get(key) ⟦OK⟧
    }

    pub fn put(&mut self, key: String, value: Vec<u8>) {
        let full = self.map.len() >= self.capacity; ⟦BUG correctness|does not check whether the key already exists, so overwriting a key in a full cache evicts an unrelated entry⟧
        if full {
            if let Some(old) = self.order.pop_back() { ⟦BUG correctness|the back of the deque is the most recently used entry; eviction must pop_front to drop the least recently used⟧
                self.map.remove(&old);
            }
        }
        self.touch(&key);
        self.map.insert(key, value);
    }

    pub fn remove(&mut self, key: &str) -> Option<Vec<u8>> {
        self.order.retain(|k| k != key);
        self.map.remove(key)
    }

    pub fn shrink_to(&mut self, new_capacity: usize) {
        self.capacity = new_capacity.max(1);
        if self.order.len() > self.capacity { ⟦BUG correctness|if evicts at most one entry; it must loop until len <= capacity, otherwise the cache stays over capacity⟧
            let old = self.order.pop_front().unwrap();
            self.map.remove(&old);
        }
    }

    pub fn len(&self) -> usize {
        self.map.len()
    }
}

=== y1-rust-05-sample-ring|Rust audio sample ring buffer|rust|src/ring.rs
CONTEXT: Fixed-size ring buffer of audio samples, used from a single thread; capacity is always at least 1. `push` overwrites the oldest sample when full, `pop` removes and returns the oldest sample, and `latest(n)` returns the n most recent samples oldest-first (n may exceed len). `rms` covers only the stored samples and is 0.0 for an empty ring.
---
pub struct Ring {
    buf: Vec<f32>,
    head: usize,
    len: usize,
}

impl Ring {
    pub fn with_capacity(cap: usize) -> Self {
        Ring { buf: vec![0.0; cap], head: 0, len: 0 }
    }

    fn cap(&self) -> usize {
        self.buf.len()
    }

    pub fn push(&mut self, sample: f32) {
        let tail = (self.head + self.len) % self.cap();
        self.buf[tail] = sample;
        if self.len == self.cap() {
            self.head = (self.head + 1) % self.cap(); ⟦OK⟧
        } else {
            self.len += 1;
        }
    }

    pub fn pop(&mut self) -> Option<f32> {
        if self.len == 0 {
            return None;
        }
        let v = self.buf[self.head];
        self.head += 1; ⟦BUG correctness|head is not wrapped modulo the capacity, so it runs past the end and the next pop or push indexes out of bounds⟧
        self.len -= 1;
        Some(v)
    }

    pub fn latest(&self, n: usize) -> Vec<f32> {
        let n = n.min(self.len);
        let start = self.head + self.len - n - 1; ⟦BUG correctness|off by one: the window starts at head + len - n; the extra - 1 returns one older sample and underflows when head is 0 and n == len⟧
        (0..n).map(|i| self.buf[(start + i) % self.cap()]).collect()
    }

    pub fn rms(&self) -> f32 {
        let sum: f32 = self.buf.iter().map(|s| s * s).sum(); ⟦BUG correctness|sums the whole backing buffer, including stale slots left by pop and unused slots, instead of only the len stored samples⟧
        let count = self.len.max(1) as f32; ⟦OK⟧
        (sum / count).sqrt()
    }

    pub fn len(&self) -> usize {
        self.len
    }

    pub fn clear(&mut self) {
        self.head = 0;
        self.len = 0;
    }
}

use std::io::{self, BufWriter, Write};
use std::path::PathBuf;

pub struct RotatingLog {
    path: PathBuf,
    max_bytes: u64,
    keep: u32,
    size: u64,
    out: BufWriter<File>,
}

fn open_append(path: &PathBuf) -> io::Result<File> {
    OpenOptions::new().create(true).append(true).open(path)
}

impl RotatingLog {
    pub fn open(path: PathBuf, max_bytes: u64, keep: u32) -> io::Result<Self> {
        let file = open_append(&path)?;
        let size = 0; ⟦BUG correctness|ignores the size of an existing file (file.metadata()?.len()), so a reopened log grows past max_bytes before rotating⟧
        Ok(RotatingLog { path, max_bytes, keep, size, out: BufWriter::new(file) })
    }

    fn rotated(&self, n: u32) -> PathBuf {
        PathBuf::from(format!("{}.{}", self.path.display(), n))
    }

    fn rotate(&mut self) -> io::Result<()> {
        self.out.flush()?;
        let _ = fs::remove_file(self.rotated(self.keep)); ⟦OK⟧
        for n in 1..self.keep { ⟦BUG data-integrity|renames run in ascending order, so .1 overwrites .2 before .2 is moved; the loop must go from keep-1 down to 1⟧

                fs::rename(&from, self.rotated(n + 1))?;
            }

        fs::rename(&self.path, self.rotated(1))?;
        self.out = BufWriter::new(open_append(&self.path)?);
        self.size = 0;
        Ok(())

use std::sync::atomic::{AtomicUsize, Ordering};
use std::sync::mpsc::{channel, Receiver, Sender};
use std::sync::{Arc, Mutex};
use std::thread::{self, JoinHandle};

type Job = Box<dyn FnOnce() + Send + 'static>;

pub struct Pool {
    tx: Option<Sender<Job>>,
    workers: Vec<JoinHandle<()>>,
    pub done: Arc<AtomicUsize>,
}

    pub fn new(n: usize) -> Pool {
        let (tx, rx) = channel::<Job>();
        let rx: Arc<Mutex<Receiver<Job>>> = Arc::new(Mutex::new(rx));

        let mut workers = Vec::with_capacity(n);
        for _ in 0..n {
            let rx = rx.clone();
            let done = done.clone();

                let job = rx.lock().unwrap().recv(); ⟦OK⟧

                    Ok(job) => {
                        job(); ⟦BUG error-handling|a panicking job unwinds and kills the worker thread; it must run under panic::catch_unwind(AssertUnwindSafe(job))⟧
                        done.fetch_add(1, Ordering::Relaxed); ⟦OK⟧

                    Err(_) => break,
                }
            }));
        }
        Pool { tx: Some(tx), workers, done }
    }

        if let Some(tx) = &self.tx {
            let _ = tx.send(Box::new(f)); ⟦BUG error-handling|the send error is discarded and true is returned even when no worker is left to receive the job⟧

            let _ = w.join();
        }
        self.tx.take();
    }

    fn drop(&mut self) {
        self.tx.take();
    }
